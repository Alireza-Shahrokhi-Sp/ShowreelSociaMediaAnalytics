import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.ticker as mticker
import json, os

# ── load data ──────────────────────────────────────────────────────────────
_pb   = pd.read_parquet(f"{LOCAL_DIR}/pathway_b_assignments_{PLATFORM}.parquet")
_pb["author_id"] = _pb["author_id"].astype(str)

_sent = pd.read_parquet(SENTIMENT_PATH, columns=["author_id", SENTIMENT_STRATA_COL])
_sent["author_id"] = _sent["author_id"].astype(str)
_dom  = (_sent.groupby("author_id")[SENTIMENT_STRATA_COL]
              .agg(lambda x: x.value_counts().idxmax())
              .reset_index()
              .rename(columns={SENTIMENT_STRATA_COL: "dominant_sentiment"}))

_cmap = json.load(open(CLUSTER_PERSONA_MAP_PATH, encoding="utf-8"))
_cmap_int = {int(k): v for k, v in _cmap.items()}

_df = (_pb.merge(_dom, on="author_id", how="left")
          .dropna(subset=["dominant_sentiment"])
          .copy())
_df = _df[_df["macro_cluster"] != -1]               # drop HDBSCAN noise
_df["persona"] = _df["macro_cluster"].map(_cmap_int).fillna("UNKNOWN")

os.makedirs(f"{LOCAL_DIR}/figs", exist_ok=True)

SENT_ORDER  = ["negative", "neutral", "positive"]
SENT_COLORS = {"negative": "#d62728", "neutral": "#aec7e8", "positive": "#2ca02c"}

# ── figure 1: persona distribution for NEGATIVE users ─────────────────────
fig1, ax1 = plt.subplots(figsize=(10, 5))
neg = (_df[_df["dominant_sentiment"] == "negative"]
       .groupby("persona").size().sort_values(ascending=False))
ax1.bar(neg.index, neg.values, color=SENT_COLORS["negative"], alpha=0.85)
ax1.set_title("Persona Distribution — Negative-Dominant Users", fontsize=13, pad=10)
ax1.set_xlabel("Persona")
ax1.set_ylabel("User count")
ax1.yaxis.set_major_locator(mticker.MaxNLocator(integer=True))
plt.xticks(rotation=35, ha="right", fontsize=9)
plt.tight_layout()
plt.savefig(f"{LOCAL_DIR}/figs/persona_negative_users.png", dpi=150)
plt.show()
print("saved persona_negative_users.png")

# ── figure 2: stacked 100% bar — sentiment mix per persona ────────────────
_ct = (_df.groupby(["persona", "dominant_sentiment"])
          .size().unstack(fill_value=0)
          .reindex(columns=SENT_ORDER, fill_value=0))
_pct = _ct.div(_ct.sum(axis=1), axis=0) * 100
_pct = _pct.loc[_pct.sum(axis=1).sort_values(ascending=False).index]

fig2, ax2 = plt.subplots(figsize=(11, 5))
bottom = pd.Series(0.0, index=_pct.index)
for sent in SENT_ORDER:
    vals = _pct[sent]
    ax2.bar(_pct.index, vals, bottom=bottom,
            label=sent.capitalize(), color=SENT_COLORS[sent], alpha=0.88)
    bottom += vals
ax2.set_title("Sentiment Mix per Persona (% of persona users)", fontsize=13, pad=10)
ax2.set_xlabel("Persona (sorted by total users)")
ax2.set_ylabel("% of users")
ax2.set_ylim(0, 100)
ax2.legend(title="Dominant sentiment", loc="upper right", fontsize=9)
plt.xticks(rotation=35, ha="right", fontsize=9)
plt.tight_layout()
plt.savefig(f"{LOCAL_DIR}/figs/persona_sentiment_mix.png", dpi=150)
plt.show()
print("saved persona_sentiment_mix.png")

# ── figure 3: heatmap — negative-user share (%) per persona ───────────────
_neg_share = (_ct["negative"] / _ct.sum(axis=1) * 100).sort_values(ascending=False)

fig3, ax3 = plt.subplots(figsize=(10, 3))
im = ax3.imshow(_neg_share.values.reshape(1, -1), aspect="auto",
                cmap="Reds", vmin=0, vmax=max(_neg_share.max(), 5))
ax3.set_xticks(range(len(_neg_share)))
ax3.set_xticklabels(_neg_share.index, rotation=35, ha="right", fontsize=9)
ax3.set_yticks([])
plt.colorbar(im, ax=ax3, orientation="horizontal", pad=0.35,
             label="% negative-dominant users")
ax3.set_title("Negative-User Concentration per Persona", fontsize=12, pad=8)
plt.tight_layout()
plt.savefig(f"{LOCAL_DIR}/figs/persona_negative_heatmap.png", dpi=150)
plt.show()
print("saved persona_negative_heatmap.png")

# ── summary table ──────────────────────────────────────────────────────────
print("\nPersona  |  total  |  neg%  |  pos%")
print("-" * 46)
for p, row in _pct.iterrows():
    total = int(_ct.loc[p].sum())
    print(f"  {p:<32s}  {total:5d}   {row['negative']:5.1f}%   {row['positive']:5.1f}%")
